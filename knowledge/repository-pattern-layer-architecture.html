// Presentation — บาง
[HttpPost]
public async Task<IActionResult> Place(PlaceOrderRequest req, CancellationToken ct)
{
    var id = await _placeOrder.Handle(req.ToCommand(), ct);
    return CreatedAtAction(nameof(Get), new { id }, null);
}

// Application — ลำดับของ use case
public async Task<int> Handle(PlaceOrderCommand cmd, CancellationToken ct)
{
    var order = Order.Create(cmd.CustomerId, cmd.Lines);
    orders.Add(order);
    await uow.SaveChangesAsync(ct);
    return order.Id;
}

// Domain — กติกาธุรกิจ
public static Order Create(int customerId, IReadOnlyList<OrderLine> lines)
{
    if (lines.Count == 0) throw new DomainException("ต้องมีสินค้าอย่างน้อย 1 รายการ");
    if (lines.Sum(l => l.Total) < 100) throw new DomainException("ยอดขั้นต่ำ 100 บาท");
    return new Order(customerId, lines);
}

หลักแบ่ง: Presentation รู้ HTTP, Application รู้ use case, Domain รู้กติกาธุรกิจ, Infrastructure รู้เทคโนโลยีภายนอกและ persistence

02Repository — ทำให้ DB ดูเหมือน collection ของ object

Application ขอข้อมูลผ่าน interface ที่สื่อความหมายของ domain โดยไม่ต้องรู้ว่า implementation ใช้ EF Core, Dapper หรือ SQL แบบใด

Interface
public interface IOrderRepository
{
    Task<Order?> GetWithLinesAsync(int id, CancellationToken ct);
    Task<IReadOnlyList<Order>> GetPendingForCustomerAsync(int customerId, CancellationToken ct);
    Task<bool> HasOpenOrderAsync(int customerId, CancellationToken ct);
    void Add(Order order);
    void Remove(Order order);
}
Infrastructure implementation
public class OrderRepository(ShopDbContext db) : IOrderRepository
{
    public Task<Order?> GetWithLinesAsync(int id, CancellationToken ct) =>
        db.Orders.Include(o => o.Lines)
          .FirstOrDefaultAsync(o => o.Id == id, ct);

    public void Add(Order order) => db.Orders.Add(order);
}
กติกาเหตุผล
1 repository ต่อ aggregateเช่น Order + OrderLine เป็น boundary เดียวกัน
ชื่อ method บอกเจตนาGetPendingForCustomer สื่อ use case ดีกว่า generic expression
ไม่คืน IQueryableไม่ปล่อย EF query API รั่วไป Application
ไม่มี SaveChanges ใน repoหนึ่ง use case อาจแตะหลาย aggregate และควร commit ครั้งเดียว
รับ CancellationTokenrequest ยกเลิก → query หยุดตาม
ไม่มี business ruleกติกาอยู่ Domain ไม่ใช่ persistence

เป้าหมาย: repository ไม่ได้มีไว้ซ่อน EF ทุก feature แต่มีไว้สร้าง persistence boundary ที่มีความหมายและทำให้ use case ไม่ผูกกับ query details

03กับดัก: Generic Repository ครอบ DbSet

ถ้าห่อ DbSet<T> ด้วย CRUD generic แล้วสุดท้ายคืน IQueryable ให้ชั้นบน abstraction จะไม่ช่วยแยก persistence จริง

ตัวอย่างที่มีปัญหา
public interface IRepository<T> where T : class
{
    Task<T?> GetByIdAsync(int id);
    Task<IEnumerable<T>> GetAllAsync();
    IQueryable<T> Query();
    Task AddAsync(T entity);
    void Update(T entity);
    void Delete(T entity);
    Task SaveAsync();
}

var orders = await _repo.Query()
    .Include(o => o.Lines)
    .Where(o => o.Status == "Pending")
    .ToListAsync();
ปัญหาผล
GetAllเปิดทางให้โหลดทั้งตารางแล้วกรองทีหลัง
Query() คืน IQueryableEF รั่วขึ้น service และ query กระจายทั่วระบบ
Update(entity)ชนกับ tracking model ของ EF และอาจ mark field เกินจำเป็น
Save ในทุก repocommit/transaction กระจาย
generic abstractionปิดความสามารถเฉพาะของ EF เช่น projection, split query, bulk update

ทางเลือก: ทำ repository เฉพาะ aggregate เมื่อ boundary มีคุณค่า หรือให้ handler ใช้ DbContext ตรงใน Vertical Slice เมื่อไม่ต้องการ abstraction เพิ่ม

04Unit of Work — commit ครั้งเดียวต่อ use case

ใน EF Core ตัว DbContext ทำหน้าที่ Unit of Work อยู่แล้ว: track การเปลี่ยนแปลงหลาย entity แล้ว SaveChangesAsync() พร้อมกัน

public interface IUnitOfWork
{
    Task<int> SaveChangesAsync(CancellationToken ct);
}

public class ShopDbContext : DbContext, IUnitOfWork { }

services.AddDbContext<ShopDbContext>(...);
services.AddScoped<IUnitOfWork>(sp =>
    sp.GetRequiredService<ShopDbContext>());
services.AddScoped<IOrderRepository, OrderRepository>();
services.AddScoped<IInventoryRepository, InventoryRepository>();
public async Task Handle(ShipOrderCommand cmd, CancellationToken ct)
{
    var order = await orders.GetWithLinesAsync(cmd.OrderId, ct)
        ?? throw new NotFoundException();
    var stock = await inventory.GetForSkusAsync(order.Skus, ct);

    order.Ship();
    stock.Deduct(order.Lines);

    await uow.SaveChangesAsync(ct); // commit เดียว
}

แนวคิด: หลาย repository ใน request เดียวควรใช้ DbContext scoped instance เดียวกัน เพื่อให้การเปลี่ยนแปลงอยู่ใน transaction boundary เดียวเมื่อเหมาะสม

05ทิศทาง Dependency — จุดต่างสำคัญของ Layered กับ Clean

จำนวนชั้นอาจเท่ากัน แต่ทิศทาง project reference และตำแหน่งของ interface ทำให้ coupling ต่างกัน

Application/Domain ประกาศ contract ส่วน Infrastructure implement ผ่าน Dependency Inversion ทำให้แกนในไม่ต้อง reference framework หรือ storage

ชื่อไม่สำคัญเท่าลูกศร: จุดสำคัญคือ business rule ไม่ควรถูก framework/persistence เป็นคนกำหนดทิศทาง dependency

06รูปแบบการแบ่งชั้น 4 แบบที่ใช้กันจริง

แต่ละแบบเหมาะกับความซับซ้อน ขนาดทีม และชนิดของ workload ที่ต่างกัน

1. Layered (N-tier)

Controller → Service → Repository
Shop.Api/Controllers Shop.Business/Services Shop.Data/Entities, Repositories, DbContext
เข้าใจง่าย เริ่มเร็ว
service โตง่ายและ feature หนึ่งแตะหลาย project

เหมาะกับ: CRUD API เล็ก–กลาง ทีมเดียว กติกาไม่ลึกมาก

2. Clean Architecture

Domain ← Application ← Infrastructure / Api
Shop.Domain/ Shop.Application/ Shop.Infrastructure/ Shop.Api/
domain test ได้โดยไม่พึ่ง EF, boundary ชัด
project/interface/boilerplate เพิ่ม

เหมาะกับ: domain ซับซ้อน ระบบอายุยาว หลายทีม

3. Vertical Slice

จัดตาม feature ไม่ใช่ technical layer
Features/ Orders/ PlaceOrder.cs ShipOrder.cs GetOrderById.cs Products/
แก้ feature อยู่ใกล้กันและใช้ EF ได้เต็มที่
code อาจซ้ำข้าม slice และมัก test ผ่าน DB จริง

เหมาะกับ: API feature เยอะ แต่แต่ละ feature ไม่ลึกมาก ทีมคุ้น EF

4. CQRS

Command ผ่าน domain · Query เน้น read model
Commands/PlaceOrder/ Queries/GetOrderList/ Infrastructure/Persistence/ Infrastructure/ReadModels/
write รักษากติกา, read optimize query ได้ตรงโจทย์
มีสอง model/flow ให้ดูแล

เหมาะกับ: read เยอะ รายงานซับซ้อน หรือ query กับ write มี requirement ต่างกันมาก

ในทางปฏิบัติ: สามารถผสมได้ เช่น Clean ฝั่ง write + query ตรงสำหรับ read หรือ Vertical Slice ที่เมื่อ logic ซับซ้อนค่อยย้ายลง Domain

08เลือกอย่างไร

CRUD เป็นหลัก ทีมเล็ก กติกาไม่ซับซ้อนVertical Slice / Layered
มีกติกาที่ต้อง test ละเอียดและมี aggregate ชัดClean + Repository เฉพาะ aggregate
รายงานและหน้ารายการเยอะ Query ซับซ้อนCQRS / Read model
หลายทีมอยู่ codebase เดียวกันระยะยาวClean + Modular Monolith
ยังไม่แน่ใจว่าจะซับซ้อนแค่ไหนเริ่มเบา แล้วเพิ่ม boundary เมื่อมีเหตุผล

สัญญาณว่าต้องเพิ่มชั้น: business rule ซ้ำหลาย handler, test ทุกอย่างต้องเปิด DB, query หลุดถึง Controller. สัญญาณว่าชั้นเกิน: interface แค่ส่งต่อ, implementation เดียวที่ไม่มี boundary value และแต่ละชั้นไม่มี logic ของตัวเอง

สรุปใจความสำคัญ

Repository, Unit of Work และ Layer Architecture มีไว้จัด dependency และความรับผิดชอบให้ชัด ไม่ใช่เพิ่มจำนวนไฟล์ให้มากที่สุด ถ้ากติกาธุรกิจอยู่ใน Domain, use case อยู่ใน Application, persistence อยู่ Infrastructure และการเปลี่ยน storage หรือ framework ไม่ทำให้ต้องแก้ business rule หลายจุด แสดงว่า boundary ของระบบกำลังทำหน้าที่ได้ดี
