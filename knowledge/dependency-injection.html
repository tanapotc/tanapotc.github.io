Dependency Injection
ใครสร้างของให้ใคร และของนั้นมีชีวิตอยู่นานแค่ไหน

DI ตอบคำถามเดียว: คลาสหนึ่งได้ของที่มันต้องใช้มาจากไหน — สร้างเองด้วย new หรือมีคนยื่นให้ทางประตูหน้า (constructor) และถ้ามีคนยื่นให้ คนนั้นคือใคร เขาสร้างของเมื่อไร และเก็บไว้ใช้ซ้ำนานแค่ไหน คำถามข้อสุดท้ายคือเรื่อง lifetime ซึ่งเป็นจุดที่พลาดกันมากที่สุดใน .NET

อ่าน 4 ขั้นนี้ให้จบ แล้วจะเห็นประโยชน์ทั้งหมดของ DI ในหน้าเดียว

ขั้น 1 ไม่มี DI — ส่งได้แค่อีเมล และเทสไม่ได้ 
public class OrderNotifier
{
    public void Notify(Order o)
    {
        // ผูกกับ SMTP ตายตัว: ส่งอีเมลจริงทุกครั้งที่เรียก method นี้
        var smtp = new SmtpClient("smtp.company.com");
        smtp.Send(o.Email, "คำสั่งซื้อสำเร็จ");
    }
}
ขั้น 2 บอกแค่ความต้องการ ไม่ระบุวิธี 
// "ใครก็ได้ที่ส่งข้อความได้" — ไม่สนว่าเป็นอีเมล, LINE หรือ SMS
public interface IMessenger
{
    Task SendAsync(string to, string text, CancellationToken ct);
}

public class OrderNotifier(IMessenger messenger)      // ← ใบสั่งของ
{
    public Task Notify(Order o, CancellationToken ct) =>
        messenger.SendAsync(o.Email, "คำสั่งซื้อสำเร็จ", ct);
}
ขั้น 3 ตัดสินใจว่า “ของจริง” คืออะไร — ที่เดียว

ขั้น 4 เทสได้โดยไม่ส่งอีเมลจริง 
public class FakeMessenger : IMessenger
{
    public List<(string To, string Text)> Sent { get; } = [];
    public Task SendAsync(string to, string text, CancellationToken ct)
    { Sent.Add((to, text)); return Task.CompletedTask; }
}

[Fact]
public async Task Notify_SendsToCustomerEmail()
{
    var fake = new FakeMessenger();
    await new OrderNotifier(fake)
        .Notify(new Order { Email = "somchai@example.com" }, default);

    fake.Sent.Should().ContainSingle(m => m.To == "somchai@example.com");
}

สังเกตว่าอะไรเปลี่ยนไปจริงๆ: โค้ดที่ทำงานยังทำงานเหมือนเดิม สิ่งที่ย้ายที่คือ ความรู้ว่า “ของจริง” คืออะไร — จากที่กระจายอยู่ในทุกคลาส กลายเป็นอยู่ที่ Program.cs ที่เดียว และเพราะมันถูกส่งเข้ามาจากข้างนอก เทสจึงส่งของปลอมเข้าไปแทนได้

ได้อะไร เสียอะไร ควรใช้เมื่อไร

ได้อะไร

  • เทสได้จริง — ยื่นของปลอมให้ ไม่ต้องมี DB / SMTP / API ภายนอก
  • สลับของได้ — เปลี่ยน SQL Server เป็น Postgres, อีเมลเป็น LINE โดยไม่แตะโค้ดธุรกิจ
  • อ่านแล้วรู้ — constructor บอกครบว่าคลาสนี้พึ่งอะไร ไม่ต้องไล่อ่านทั้งไฟล์
  • จัดการอายุของ object ให้ — เปิด/ปิด connection, dispose ตอนจบ request ให้อัตโนมัติ
  • ประกอบที่เดียว — อยากรู้ว่าระบบใช้อะไรอยู่ เปิดไฟล์เดียวจบ

เสียอะไร

  • อ้อมขึ้น — เห็น IMessenger แล้วต้องไปดู Program.cs ว่าตัวจริงคืออะไร (IDE กด F12 ไปไม่ถึงทันที)
  • ไฟล์เยอะขึ้น — ได้ interface เพิ่มมาต่อหนึ่ง service
  • พังตอน runtime ได้ — ลืม register แล้ว compiler ไม่เตือน (แก้ด้วย ValidateOnBuild — หัวข้อ 3)
  • เรื่อง lifetime เป็นกับดักใหม่ — เลือกผิดแล้วข้อมูลรั่วข้าม request หรือ DbContext บวม (หัวข้อ 4–5)
  • ถูกใช้เกิน — ใส่ interface ให้ทุกคลาสทั้งที่มี implementation เดียวและไม่เคย mock

ควรใช้เมื่อไร

ควรใช้ — เกือบทุกกรณี
  • ของชิ้นนั้นแตะ โลกภายนอก: DB, HTTP, ไฟล์, อีเมล, เวลา, สุ่ม
  • อยากเขียนเทสให้ logic ที่เรียกใช้มัน
  • มีหลาย implementation หรืออาจมีในอนาคต
  • ของสร้างแพงและต้องใช้ซ้ำ (connection pool, HttpClient, cache)
ไม่ต้องใช้
  • Value object / DTO / entity — new Order(...) ได้เลย ไม่ต้องผ่าน container
  • ฟังก์ชันคำนวณล้วนที่ไม่แตะอะไรข้างนอก (static ก็พอ)
  • คลาสที่มี implementation เดียว ไม่เคยสลับ และไม่เคย mock — new ตรงๆ อ่านง่ายกว่า
สำหรับ .NET โดยเฉพาะ: DI ไม่ใช่ทางเลือก — ASP.NET Core สร้างบนมันทั้งหมด ILogger, IConfiguration, DbContext, controller, middleware ทุกตัวถูกส่งเข้ามาผ่าน container อยู่แล้ว คำถามจึงไม่ใช่ “จะใช้ DI ไหม” แต่เป็น “จะใช้ให้ถูกไหม” — โดยเฉพาะเรื่อง lifetime ที่เป็นหัวข้อ 4 ถึง 6 ของหน้านี้

01ปัญหาของการ new เอง

โค้ดสองชุดนี้ทำงานเหมือนกันทุกอย่าง ต่างกันแค่ว่า OrderService รู้จัก "ของจริง" หรือรู้จักแค่ "สัญญา"

ผูกกันแน่น — สร้างเองข้างใน 
public class OrderService
{
    private readonly SqlOrderRepository _repo;
    private readonly SmtpEmailSender _email;

    public OrderService()
    {
        _repo  = new SqlOrderRepository(
            "Server=prod-db;User=sa;Password=...");   // ← connection string ฝังในโค้ด
        _email = new SmtpEmailSender("smtp.company.com", 587);
    }

    public async Task Place(Order o)
    {
        await _repo.SaveAsync(o);
        await _email.SendAsync(o.Email, "ยืนยันคำสั่งซื้อ");
    }
}
เทสไม่ได้เรียก Place() ในเทส = ยิง DB จริงและส่งอีเมลจริง
เปลี่ยนไม่ได้อยากสลับไป PostgresOrderRepository หรือส่งผ่าน LINE ต้องแก้ข้างใน OrderService
ซ่อนของที่ต้องใช้อ่าน constructor แล้วไม่รู้ว่าคลาสนี้พึ่งอะไรอยู่ — ต้องไล่อ่านทั้งไฟล์
เปลืองทรัพยากรทุกครั้งที่ new OrderService() จะเปิด connection / SMTP client ใหม่
ผูกกันหลวม — รับเข้ามาทางประตูหน้า 
// รู้จักแค่ interface ไม่รู้ว่าเบื้องหลังเป็น SQL, Postgres หรือ fake
public class OrderService(
    IOrderRepository repo,
    IEmailSender email,
    ILogger<OrderService> log)          // primary constructor (C# 12)
{
    public async Task Place(Order o, CancellationToken ct)
    {
        await repo.SaveAsync(o, ct);
        await email.SendAsync(o.Email, "ยืนยันคำสั่งซื้อ", ct);
        log.LogInformation("placed order {Id}", o.Id);
    }
}

// Program.cs — ที่เดียวที่รู้ว่า "ของจริง" คืออะไร
builder.Services.AddScoped<IOrderRepository, SqlOrderRepository>();
builder.Services.AddScoped<IEmailSender, SmtpEmailSender>();
builder.Services.AddScoped<OrderService>();

สิ่งที่เปลี่ยนไปจริงๆ: ความรู้ว่า "ของจริงคืออะไร" ถูกย้ายออกจากคลาสที่ใช้งาน ไปอยู่ที่ จุดเดียว ตอนสตาร์ตแอป — OrderService เหลือหน้าที่แค่ "ทำงานตามลำดับ" ซึ่งเป็นสิ่งที่มันควรรู้จริงๆ

02หลักการ — คำสามคำที่มักปนกัน

Dependency Injectionเทคนิค — ส่งของที่ต้องใช้เข้ามาจากข้างนอก แทนที่จะสร้างเองข้างใน (ทำมือก็ได้ ไม่ต้องมี container)
Inversion of Controlหลักการกว้างๆ — ยกหน้าที่ควบคุม (สร้างของ, เรียกเมื่อไร) ไปให้ framework แทนที่โค้ดเราจะคุมทุกอย่าง DI เป็น IoC แบบหนึ่ง
Dependency Inversion (D ใน SOLID)ทิศทางของ reference — ชั้นบนกับชั้นล่างต่างพึ่ง abstraction ไม่ให้ชั้นบนพึ่งชั้นล่างตรงๆ (นี่คือเหตุผลที่ interface อยู่ชั้น Application ไม่ใช่ Infrastructure)
DI Container / IoC Containerเครื่องมือ — ตัวที่จำว่า interface ไหนคู่กับ class ไหน แล้วประกอบ object ให้อัตโนมัติ ใน .NET คือ IServiceProvider

3 ช่องทางที่ inject ได้

แบบใช้เมื่อ
Constructorค่าเริ่มต้นเกือบ 100% — ของที่ขาดไม่ได้ ประกาศเป็น readonly ได้ และถ้าลืม register แอปจะพังตอนสตาร์ต ไม่ใช่ตอน runtime
Method / parameterของที่ใช้แค่ method เดียว — Minimal API และ [FromServices] ใน MVC ใช้แบบนี้
Property / setterของที่มีค่า default ใช้ได้อยู่แล้ว (เช่น ILogger ที่ default เป็น NullLogger) — .NET container ในตัวไม่รองรับ ต้องทำเองหรือใช้ container ตัวอื่น เลี่ยงไว้ก่อน เพราะซ่อนว่าคลาสพึ่งอะไร

กฎข้อเดียวที่จำง่าย: constructor ของคลาสคือ "ใบแจ้งรายการ" ว่าคลาสนี้ทำงานไม่ได้ถ้าขาดอะไร — อ่าน constructor แล้วต้องรู้ทันทีว่ามันพึ่งอะไรอยู่ ถ้าต้องไปหาในเนื้อ method แปลว่าซ่อนของแล้ว

03Container สร้างของอย่างไร — object graph

ตอน request เข้ามา container ไม่ได้สร้างแค่ controller มันไล่ดูว่า controller ขออะไร แล้วของนั้นขออะไรต่อ ลงไปเรื่อยๆ จนถึงตัวที่ไม่ขออะไรอีก แล้วประกอบย้อนขึ้นมา กดดูทีละก้าว

// สิ่งที่เราเขียน — ไม่มีใคร new ใครเลย
public class OrdersController(OrderService svc) : ControllerBase;

public class OrderService(IOrderRepository repo, IEmailSender email);

public class SqlOrderRepository(ShopDbContext db) : IOrderRepository;

public class SmtpEmailSender(IOptions<SmtpOptions> opt) : IEmailSender;

// สิ่งที่ container ทำแทนเรา (เทียบเท่าโค้ดนี้)
var db      = new ShopDbContext(options);
var repo    = new SqlOrderRepository(db);
var email   = new SmtpEmailSender(smtpOptions);
var svc     = new OrderService(repo, email);
var ctrl    = new OrdersController(svc);
ลำดับการสร้างลึกสุดก่อน แล้วประกอบย้อนขึ้น — เพราะต้องมีของครบก่อนจะ new ตัวที่ต้องใช้มันได้
ถ้าลืม registerInvalidOperationException: Unable to resolve service for type '...' — เกิดตอนที่ container พยายามสร้างชั้นนั้น
Circular dependencyA ขอ B, B ขอ A → container โยน exception ตอน resolve เป็นสัญญาณว่า design ผิด ให้แยกส่วนที่ทั้งคู่ต้องใช้ออกเป็นตัวที่สาม
ตรวจล่วงหน้าได้ใน Development .NET เปิด ValidateOnBuild + ValidateScopes ให้ — พังตอนสตาร์ต ดีกว่าพังตอนผู้ใช้กด

วิธีคิดที่ใช้ตัดสินใจได้เร็ว: ถามว่า "ของชิ้นนี้ผูกกับอะไร" — ผูกกับ คำขอหนึ่งครั้ง (มี transaction, มี user ปัจจุบัน) → Scoped · ผูกกับ ทั้งแอป (ไม่มี state, สร้างแพง) → Singleton · ไม่ผูกกับอะไรเลย (สร้างเมื่อไรก็เหมือนกัน) → Transient

05Captive dependency — bug ที่ compiler จับไม่ได้

ถ้าตัวที่อายุยืนถือตัวที่อายุสั้นไว้ ตัวอายุสั้นจะถูก "กักขัง" ให้อยู่ยาวตามคนที่ถือมัน — DbContext ที่ควรตายพร้อม request กลายเป็นอยู่ตลอดอายุแอป

});

06Background job — กรณีที่เจอ captive dependency บ่อยที่สุด

BackgroundService ถูก register เป็น Singleton เพราะมันรันตลอดอายุแอป ดังนั้นจะ inject DbContext หรือ repository เข้า constructor ตรงๆ ไม่ได้ ต้องสร้าง scope ขึ้นมาเองทุกรอบงาน

พังตอนสตาร์ต (หรือแย่กว่า: ไม่พังแต่ข้อมูลเพี้ยน) 
public class OrderSyncWorker(ShopDbContext db)   // ← Scoped เข้า Singleton
    : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        while (!ct.IsCancellationRequested)
        {
            var pending = await db.Orders
                .Where(o => o.Status == OrderStatus.Pending)
                .ToListAsync(ct);
            // ... db ตัวเดียวนี้จะอยู่กับ worker ตลอดอายุแอป
            await Task.Delay(TimeSpan.FromMinutes(1), ct);
        }
    }
}

ถ้าเปิด ValidateScopes จะพังตอนสตาร์ตทันที (ซึ่งดี) ถ้าไม่เปิด มันจะรันได้ แล้วค่อยๆ เจอ: change tracker โตขึ้นทุกนาที, ข้อมูลที่อ่านมาเป็นของเก่าที่ถูก cache ไว้, และ connection ค้าง

ถูก — 1 รอบงาน = 1 scope 
public class OrderSyncWorker(
    IServiceScopeFactory scopeFactory,          // Singleton — inject ได้
    ILogger<OrderSyncWorker> log) : BackgroundService
{
    protected override async Task ExecuteAsync(CancellationToken ct)
    {
        using var timer = new PeriodicTimer(TimeSpan.FromMinutes(1));
        while (await timer.WaitForNextTickAsync(ct))
        {
            try
            {
                // scope ใหม่ทุกรอบ = DbContext สะอาดทุกรอบ เหมือน request หนึ่งครั้ง
                await using var scope = scopeFactory.CreateAsyncScope();
                var svc = scope.ServiceProvider
                               .GetRequiredService<IOrderSyncService>();
                await svc.SyncPendingAsync(ct);
            }
            catch (Exception ex) when (!ct.IsCancellationRequested)
            {
                log.LogError(ex, "sync รอบนี้ล้ม — รอบหน้าลองใหม่");
                // ★ ห้ามให้ exception หลุดออกจาก ExecuteAsync
                //   ไม่งั้น worker ตายเงียบและไม่มีใครรู้
            }
        }
    }
}
ทำไมต้อง scope ต่อรอบเพื่อให้ DbContext, repository และ transaction ของรอบนั้นเกิดและตายพร้อมกัน — เหมือน request หนึ่งครั้ง
CreateAsyncScopeใช้กับ service ที่เป็น IAsyncDisposable (DbContext เป็น) — ได้ dispose แบบ async ถูกต้อง
Hangfire / Quartzทั้งสองตัวสร้าง scope ต่อ job ให้อยู่แล้ว จึง inject Scoped เข้า job class ได้ตรงๆ — ต่างจาก BackgroundService ดิบ
หลาย instanceถ้าแอปรันหลาย pod งานเดียวกันจะถูกหยิบซ้ำ — ต้องมี distributed lock หรือให้ job ทำงานแบบ idempotent

07กับดักอื่นที่เจอบ่อย

อาการ / สิ่งที่เห็นปัญหาจริงทางแก้
Constructor รับ 10 ตัวไม่ใช่ปัญหาของ DI — เป็นสัญญาณว่าคลาสนี้ทำหลายหน้าที่ (ละเมิด SRP)แตกคลาสตามหน้าที่ · หรือรวมของที่มาด้วยกันเสมอเป็น facade เดียว — ไม่ใช่เปลี่ยนไปใช้ service locator เพื่อซ่อนจำนวน
provider.GetService<T>() กระจายในโค้ดธุรกิจService Locator — กลับไปซ่อน dependency อีกครั้ง คลาสพึ่งอะไรดูจาก constructor ไม่ได้ และ mock ยากรับเข้ามาทาง constructor · ยอมรับได้เฉพาะที่ขอบระบบ (composition root, middleware, factory, background worker)
ต้องเลือก implementation ตอน runtimeเช่นเลือกช่องทางชำระเงินตามข้อมูลที่ผู้ใช้ส่งมาinject factory (Func<string, IPaymentGateway>) หรือ inject IEnumerable<IPaymentGateway> ทั้งชุดแล้วเลือกเอง หรือใช้ keyed services (หัวข้อ 9)
Register interface เดียวหลาย classคนส่วนใหญ่คิดว่าตัวหลังทับตัวหน้าจริงคือ เก็บทั้งหมด — ขอเป็น T ได้ตัวที่ register ล่าสุด ขอเป็น IEnumerable<T> ได้ทั้งหมด · ถ้าตั้งใจจะทับใช้ Replace() หรือ TryAdd*
ต้องใช้ await ตอนสร้าง servicecontainer ในตัวของ .NET สร้าง object แบบ sync เท่านั้นconstructor ต้องเบาและไม่ทำ I/O · โหลดข้อมูลตั้งต้นด้วย IHostedService ตอนสตาร์ต หรือ lazy-init ครั้งแรกที่ใช้ (AsyncLazy/SemaphoreSlim)
Constructor ทำงานหนัก / เรียก DBทุก request ช้าลง และ exception ตอนสร้างจะอ่านยากมากconstructor แค่รับของมาเก็บ — งานจริงไปอยู่ใน method
new HttpClient() เองsocket exhaustion และไม่รู้เรื่อง DNS ที่เปลี่ยนAddHttpClient<T>() แล้ว inject typed client — factory จัดการ pooling และ handler lifetime ให้
Register class เป็น concrete type ทุกตัวไม่ผิด แต่ถ้าไม่มี interface ก็ mock ในเทสไม่ได้ใส่ interface เฉพาะจุดที่ต้องสลับหรือต้อง mock จริง — ไม่ต้องมี IThing ให้ทุกคลาส
Dispose ของเองเรียก Dispose() ของ service ที่ container สร้างอย่าทำ — container เป็นเจ้าของอายุของมัน จะ dispose ให้ตอนจบ scope (ยกเว้น instance ที่เรา new แล้วส่งให้ container ด้วย AddSingleton(instance) — อันนั้นเราดูแลเอง)

08ผลพลอยได้ที่ใหญ่ที่สุด — เทสง่ายขึ้น

นี่คือเหตุผลที่จับต้องได้ที่สุดของ DI: เมื่อคลาสรับของเข้ามาทาง constructor เทสก็แค่ยื่นของปลอมให้ — ไม่ต้องมี DB, ไม่ต้องมี SMTP, ไม่ต้องมี container

// Unit test — ไม่มี DI container, ไม่มี I/O
[Fact]
public async Task Place_WhenSaved_SendsConfirmationEmail()
{
    // Arrange — ยื่นของปลอมเข้าไปตรงๆ
    var repo  = new FakeOrderRepository();
    var email = Substitute.For<IEmailSender>();
    var svc   = new OrderService(repo, email, NullLogger<OrderService>.Instance);

    // Act
    await svc.Place(new Order { Id = 1, Email = "a@b.com" }, default);

    // Assert
    repo.Saved.Should().ContainSingle(o => o.Id == 1);
    await email.Received(1).SendAsync("a@b.com", Arg.Any<string>(), Arg.Any<CancellationToken>());
}

และเมื่อต้องเทสทั้งเส้น (integration test) ก็สลับของจริงบางตัวออกที่ระดับ container โดยไม่แตะโค้ดธุรกิจเลย

// Integration test — ใช้ container จริง แต่สลับ email เป็นของปลอม
public class ApiFactory : WebApplicationFactory<Program>

ข้อสังเกต: ถ้าเขียนเทสแล้วต้องพยายามอย่างมากเพื่อ mock อะไรบางอย่าง (ต้อง mock static, ต้องตั้ง container ทั้งชุด, ต้องแก้ internal ให้เป็น public) นั่นมักเป็นสัญญาณว่า dependency ตัวนั้นยังไม่ได้ถูก inject เข้ามา — ไม่ใช่ปัญหาของเทส

จัดระเบียบ Program.cs

// แยกตามชั้น — ต่อกับหน้า layer-architecture
builder.Services
    .AddApplication()                          // use case / handler
    .AddInfrastructure(builder.Configuration);  // DbContext, repository, HTTP client

// Shop.Infrastructure/DependencyInjection.cs
public static IServiceCollection AddInfrastructure(
    this IServiceCollection s, IConfiguration cfg)
{
    s.AddDbContext<ShopDbContext>(o =>
        o.UseSqlServer(cfg.GetConnectionString("Shop")));      // Scoped
    s.AddScoped<IUnitOfWork>(sp => sp.GetRequiredService<ShopDbContext>());
    s.AddScoped<IOrderRepository, SqlOrderRepository>();
    s.AddSingleton<IClock, SystemClock>();                        // ไม่มี state
    s.AddHttpClient<IShippingApi, ShippingApi>();                 // typed client
    return s;
}

ข้อควรรู้เรื่อง TryAdd: ไลบรารีที่เขียนให้คนอื่นใช้ควร register ด้วย TryAddScoped / TryAddSingleton เพื่อไม่ทับของที่ผู้ใช้ตั้งเอง — ส่วนโค้ดในแอปของเราใช้ Add* ธรรมดาได้

Checklist

DI ไม่ใช่เรื่องของ container — container เป็นแค่เครื่องมือ สิ่งที่ DI ให้จริงๆ คือ "คลาสนี้พึ่งอะไร" กลายเป็นสิ่งที่อ่านออกจาก constructor และ "ของจริงคืออะไร" ถูกตัดสินที่จุดเดียวตอนสตาร์ตแอป · ส่วน lifetime คือคำถามว่าของแต่ละชิ้นผูกกับ request, กับทั้งแอป, หรือไม่ผูกกับอะไรเลย
