04สี่วิธีเขียน — โค้ดจริงใน C#

1. Recursion — สั้นที่สุด อันตรายที่สุด 
void Fill(int[,] g, int x, int y, int target, int newColor)
{
    if (x < 0 || y < 0 || x >= W || y >= H) return;
    if (g[y, x] != target) return;      // กำแพง/สีอื่น/เททับแล้ว

    g[y, x] = newColor;                   // ทาก่อน = ทำหน้าที่ "visited" ไปในตัว

    Fill(g, x + 1, y, target, newColor);
    Fill(g, x - 1, y, target, newColor);
    Fill(g, x, y + 1, target, newColor);
    Fill(g, x, y - 1, target, newColor);
}

// ★ ผู้เรียกต้องเช็คก่อน ไม่งั้นลูปไม่รู้จบ
if (target != newColor) Fill(g, sx, sy, target, newColor);

อ่านง่ายที่สุดและตรงกับนิยามที่สุด แต่ภาพ 1920×1080 ที่เป็นสีเดียวทั้งภาพจะเรียกซ้อนกันได้ถึงหลักล้านชั้น → StackOverflowException ซึ่งใน .NET ดักด้วย try/catch ไม่ได้ process ตายทันที ใช้ได้เฉพาะตอนที่รู้แน่ว่าพื้นที่เล็ก

3. DFS แบบสแตกของตัวเอง 
void FillDfs(int[,] g, int sx, int sy, int newColor)
{
    int target = g[sy, sx];
    if (target == newColor) return;

    var st = new Stack<(int x, int y)>();
    st.Push((sx, sy));
    g[sy, sx] = newColor;

    while (st.Count > 0)
    {
        var (x, y) = st.Pop();            // ← ต่างจาก BFS แค่บรรทัดนี้
        foreach (var (dx, dy) in Dirs4)
        {
            int nx = x + dx, ny = y + dy;
            if ((uint)nx >= W || (uint)ny >= H) continue;
            if (g[ny, nx] != target) continue;
            g[ny, nx] = newColor;         // ★ ทาตอน "ใส่เข้าสแตก" ไม่ใช่ตอนหยิบออก
            st.Push((nx, ny));
        }
    }
}

ทาสีตอนใส่เข้า container ไม่ใช่ตอนหยิบออก — ถ้าทาตอนหยิบออก ช่องเดียวจะถูกใส่ซ้ำหลายรอบจากเพื่อนบ้านหลายตัว สแตกบวมเป็นหลายเท่า

2. BFS แบบคิว 
static readonly (int dx, int dy)[] Dirs4 = [(1,0), (-1,0), (0,1), (0,-1)];

void FillBfs(int[,] g, int sx, int sy, int newColor)
{
    int target = g[sy, sx];
    if (target == newColor) return;

    var q = new Queue<(int x, int y)>();
    q.Enqueue((sx, sy));
    g[sy, sx] = newColor;

    while (q.Count > 0)
    {
        var (x, y) = q.Dequeue();
        foreach (var (dx, dy) in Dirs4)
        {
            int nx = x + dx, ny = y + dy;
            if ((uint)nx >= W || (uint)ny >= H) continue;
            if (g[ny, nx] != target) continue;
            g[ny, nx] = newColor;
            q.Enqueue((nx, ny));
        }
    }
}

(uint)nx >= W เป็นทริกเช็คขอบเขตสองด้านด้วยการเทียบครั้งเดียว — ค่าติดลบจะกลายเป็นเลขบวกมหาศาลเมื่อ cast เป็น uint จึงเข้าเงื่อนไขเดียวกัน

4. Scanline — เร็วที่สุดในทางปฏิบัติ 
void FillScanline(int[,] g, int sx, int sy, int newColor)
{
    int target = g[sy, sx];
    if (target == newColor) return;

    var st = new Stack<(int x, int y)>();
    st.Push((sx, sy));

    while (st.Count > 0)
    {
        var (sx0, y) = st.Pop();
        int x = sx0;
        while (x >= 0 && g[y, x] == target) x--;   // ถอยไปสุดซ้ายของช่วง
        x++;

        bool up = false, down = false;
        while (x < W && g[y, x] == target)        // กวาดขวายาวรวดเดียว
        {
            g[y, x] = newColor;

            // ใส่ seed แค่ "ช่องแรกของแต่ละช่วงใหม่" ไม่ใช่ทุกช่อง
            if (y > 0)
            {
                bool m = g[y - 1, x] == target;
                if (m && !up) { st.Push((x, y - 1)); up = true; }
                else if (!m) up = false;
            }
            if (y < H - 1)
            {
                bool m = g[y + 1, x] == target;
                if (m && !down) { st.Push((x, y + 1)); down = true; }
                else if (!m) down = false;
            }
            x++;
        }
    }
}

แทนที่จะจัดการทีละช่อง มันกวาดทีละ “ช่วงในแถว” และใส่ seed เฉพาะช่องแรกของช่วงใหม่ในแถวบน/ล่าง — สแตกจึงเล็กกว่ามาก และเข้าถึงหน่วยความจำต่อเนื่องตามแถว (cache-friendly) นี่คือวิธีที่โปรแกรมวาดรูปจริงใช้

อัลกอริทึมช่องที่เทจำนวนก้าวหน่วยความจำสูงสุดลักษณะการลาม

07Tolerance — จากตารางสู่ภาพจริง

ในภาพถ่ายจริงไม่มีสองพิกเซลที่สีเท่ากันเป๊ะ เงื่อนไข “สี == target” จึงต้องกลายเป็น “สีต่างจาก target ไม่เกินค่าที่ยอมรับ” ลากค่า tolerance ดูว่าช่องไหนถูกนับว่า “สีเดียวกัน”

tolerance 30
static bool IsSimilar(Rgb a, Rgb b, int tolerance)
{
    int dr = a.R - b.R, dg = a.G - b.G, db = a.B - b.B;
    return dr * dr + dg * dg + db * db <= tolerance * tolerance;
    // เทียบกำลังสอง เลี่ยง Math.Sqrt ที่เรียกหลายล้านครั้ง
}
tolerance = 0ต้องเท่ากันเป๊ะ — ใช้ได้กับภาพ pixel art / ตารางเกม
ต่ำเกินไปเหลือขอบสีเดิมรอบพื้นที่ (ผลของ anti-aliasing)
สูงเกินไปสีรั่วข้ามขอบที่ตาเห็นว่าต่างกันชัดเจน
เทียบยังไงดีกว่า RGBระยะใน RGB ไม่ตรงกับที่ตาคนรับรู้ — งานที่ต้องการคุณภาพใช้ LAB หรือ ΔE แทน
เทียบกับอะไรสีของ seed เสมอ ไม่ใช่สีของช่องก่อนหน้า — ถ้าเทียบกับช่องก่อนหน้า ค่าจะค่อยๆ เลื่อนไปเรื่อยจนเทลามทั้งภาพ (ปัญหา color drift)

มองอีกมุม: flood fill คือ BFS/DFS บนกราฟที่ “โหนด” คือช่อง และ “เส้นเชื่อม” คือความเป็นเพื่อนบ้านที่สีตรงกัน — เข้าใจอันนี้แล้วจะเห็นว่าเป็นอัลกอริทึมเดียวกับการหาเส้นทางในเขาวงกต ต่างแค่เงื่อนไขว่าเดินต่อได้เมื่อไร
